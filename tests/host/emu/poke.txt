# The gamepad tester draws whatever is in its XRAM block. This script has no
# pad command, so nothing rewrites the block after the program maps it, and a
# poked record stays.

run 40
expect "Disconnected"
expect-not "Select"

# $80 in the first byte is the connected bit and $08 in the fourth is Start.
poke xram:$FF00 $80 $00 $00 $08
run 20
expect "Select"
peek xram:$FF00 $80 $00 $00 $08

poke $0200 $12 34 0x56
peek ram:$0200 18 $22 86
