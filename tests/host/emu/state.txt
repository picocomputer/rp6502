run 40
expect "Disconnected"

mark
state save "machine.sst"

poke xram:GAMEPAD $80 $00 $00 $08
run 20
expect "Select"
expect-changed
peek xram:GAMEPAD $80 $00 $00 $08

state load "machine.sst"
run 1
expect-same
peek xram:GAMEPAD $00 $00 $00 $00

poke $0200 $52 $50 $36 $34

# A machine saved while it is stopped loads back.
sys stop
state save "stopped.sst"
state load "stopped.sst"
peek $0200 $52 $50 $36 $34
