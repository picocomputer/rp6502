# mouse.rp6502 maps the mouse block at $FF00. Its 6522 timer interrupt copies
# the block's five bytes to $FF50 and adds one to a count at $FF55. Each wait
# is on the copy at $FF50, so it passes only after an interrupt has read the
# new values.

run 10
peek xram:$FF00 $00 $00 $00 $00 $00

# Timer 1 free-runs from $FFFF, so it interrupts about every 8 ms at the
# default 8 MHz clock and ten 60 Hz frames hold twenty interrupts. The count
# starts from zero because --fill 0 clears XRAM at boot.
peek xram:$FF55 $14

mouse buttons 1
wait xram:$FF50 $01
peek xram:$FF00 $01 $00 $00 $00 $00

mouse move 60 40
wait xram:$FF51 $3C
peek xram:$FF00 $01 $3C $28
peek xram:$FF50 $01 $3C $28

mouse move 60 40
wait xram:$FF51 $78
peek xram:$FF50 $01 $78 $50

mouse wheel 3
wait xram:$FF53 $03
peek xram:$FF00 $01 $78 $50 $03 $00

mouse buttons 0
wait xram:$FF50 $00
peek xram:$FF50 $00 $78 $50 $03 $00
