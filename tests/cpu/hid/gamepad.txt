peek xram:$FF00 $80 $00 $00 $08

pad 0 release start
pad 0 press up a
run 10
peek xram:$FF00 $81 $00 $01 $00

# Each player's record is ten bytes, so player 1's starts at $FF0A.
peek xram:$FF0A $00

# The sticks byte holds the left stick's direction in its low nibble and the
# right stick's in its high nibble, and up is bit 0 of each nibble, so the byte
# reads $01 when the left stick is pushed fully up and the right stick is
# centered.
pad 0 stick 0 -127 0 0
run 10
peek xram:$FF00 $81 $01

pad 0 disconnect
run 20
expect "Disconnected"
peek xram:$FF00 $00 $00 $00 $00 $00 $00 $00 $00 $00 $00
