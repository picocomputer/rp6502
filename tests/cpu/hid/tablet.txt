# tablet.rp6502 maps the tablet block at $FF10. The block is a four-byte
# header followed by eight contacts of six bytes: flags, x0, x1, x2, y0, y1.
# Each axis is split across single-byte windows, and only the window that holds
# the coordinate is non-zero.

run 10
peek xram:$FF14 $00 $00 $00 $00 $00 $00

# A single touch is contact 0 with its flags set to $01, TABLET_FLAG_LEFT. A
# coordinate up to 254 is in the first window, which holds the coordinate plus
# one.
tablet touch 60,60
wait xram:$FF14 $01
peek xram:$FF14 $01 $3D $00 $00 $3D $00

# A coordinate from 255 to 509 is in the second window, which holds the
# coordinate minus 254.
tablet touch 300,200
wait xram:$FF16 $2E
peek xram:$FF14 $01 $00 $2E $00 $C9 $00

tablet touch 100,300
wait xram:$FF18 $00
peek xram:$FF14 $01 $65 $00 $00 $00 $2E

# A hovering pointer is also contact 0, and its flags are its button bitmap
# with TABLET_FLAG_HOVER, $80, set.
tablet at 100 100 1
wait xram:$FF14 $81
peek xram:$FF14 $81 $65 $00 $00 $65 $00

tablet clear
wait xram:$FF14 $00
peek xram:$FF14 $00 $00 $00 $00 $00 $00

tablet touch 10,10 20,20 30,30
wait xram:$FF14 $01
peek xram:$FF14 $01 $0B $00 $00 $0B $00
peek xram:$FF1A $01 $15 $00 $00 $15 $00
peek xram:$FF20 $01 $1F $00 $00 $1F $00
